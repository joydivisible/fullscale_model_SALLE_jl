function fit_tau_nrtl(tie_lines)
    lb = fill(-2.0, 6)
    ub = fill(2.0, 6)

    # Use a smaller number of initial samples to speed up the process
    num_initial_samples = 10 
    println("PHASE 1: Building surrogate model with $num_initial_samples initial points...")
    
    # A counter to pass to the loss function for clear logging
    iter_counter = Ref(0)
    f = p -> begin
        iter_counter[] += 1
        loss_from_tau_vector(collect(p), tie_lines, iter_counter[])
    end

    X_tuples = sample(num_initial_samples, lb, ub, SobolSample())
    X = [collect(x) for x in X_tuples]
    Y = [f(x) for x in X]
    
    println("\nPHASE 2: Optimizing with surrogate model (SRBF)...")
    surrogate = Kriging(X, Y, lb, ub)
    
    # Reduce maxiters for the surrogate optimization part
    num_surrogate_steps = 10
    surrogate_optimize(f, SRBF(), lb, ub, surrogate, RandomSample(); maxiters=num_surrogate_steps)

    best_params = surrogate.x[argmin(surrogate.y)]
    best_loss_surrogate = minimum(surrogate.y)
    println("\n--- Surrogate Optimization Complete ---")
    println("Best params from surrogate: $(round.(best_params, digits=4))")
    println("Best loss from surrogate: $(round(best_loss_surrogate, digits=5))")

    # PHASE 3: Polishing the result with a gradient-based optimizer
    println("\nPHASE 3: Polishing final parameters with L-BFGS...")
    optf = OptimizationFunction((p, _) -> loss_from_tau_vector(p, tie_lines, iter_counter[]), Optimization.AutoZygote())
    prob = OptimizationProblem(optf, best_params, lb=lb, ub=ub)
    
    # The L-BFGS solver is often much faster for the final steps
    result = solve(prob, LBFGS(), maxiters=20)
    final_params = result.u
    final_loss = result.minimum
    
    tau_matrix = reshape_tau(final_params)
    return tau_matrix, final_loss
end
